cmake_minimum_required(VERSION 3.25)
project(smac_native VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SMAC_BUILD_CLIENT "Build the SDL client when SDL3 is available" ON)
option(SMAC_FETCH_DEPENDENCIES "Download pinned SDL dependencies" OFF)
option(SMAC_ENABLE_SANITIZERS "Enable address and undefined-behaviour sanitizers" OFF)

list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
include(ProjectWarnings)

add_library(smac_core
  src/core/world_map.cpp src/core/game_state.cpp src/core/pathfinding.cpp)
target_include_directories(smac_core PUBLIC include)
smac_set_warnings(smac_core)

add_library(smac_formats
  src/formats/text.cpp src/formats/data_directory.cpp src/formats/rules.cpp
  src/formats/terran_map.cpp src/formats/sha256.cpp)
target_include_directories(smac_formats PUBLIC include)
target_link_libraries(smac_formats PUBLIC smac_core)
smac_set_warnings(smac_formats)

add_executable(smac-tool src/tool/main.cpp)
target_link_libraries(smac-tool PRIVATE smac_formats)
smac_set_warnings(smac-tool)

set(SMAC_HAVE_SDL OFF)
if(SMAC_BUILD_CLIENT)
  find_package(SDL3 CONFIG QUIET)
  find_package(SDL3_image CONFIG QUIET)
  find_package(SDL3_ttf CONFIG QUIET)
  if(NOT SDL3_FOUND AND SMAC_FETCH_DEPENDENCIES)
    include(FetchContent)
    set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
    set(SDL_TESTS OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(SDL3 URL https://github.com/libsdl-org/SDL/releases/download/release-3.4.14/SDL3-3.4.14.tar.gz)
    FetchContent_Declare(SDL3_image URL https://github.com/libsdl-org/SDL_image/releases/download/release-3.4.4/SDL3_image-3.4.4.tar.gz)
    FetchContent_Declare(SDL3_ttf URL https://github.com/libsdl-org/SDL_ttf/releases/download/release-3.2.2/SDL3_ttf-3.2.2.tar.gz)
    FetchContent_MakeAvailable(SDL3 SDL3_image SDL3_ttf)
  endif()
  if(TARGET SDL3::SDL3 AND TARGET SDL3_image::SDL3_image AND TARGET SDL3_ttf::SDL3_ttf)
    set(SMAC_HAVE_SDL ON)
  endif()
endif()

if(SMAC_HAVE_SDL)
  add_executable(smac-native src/client/sdl_main.cpp)
  target_link_libraries(smac-native PRIVATE smac_formats SDL3::SDL3 SDL3_image::SDL3_image SDL3_ttf::SDL3_ttf)
else()
  add_executable(smac-native src/client/stub_main.cpp)
  target_link_libraries(smac-native PRIVATE smac_formats)
  message(STATUS "SDL3 not found: smac-native will explain how to enable the graphical client")
endif()
smac_set_warnings(smac-native)

include(CTest)
if(BUILD_TESTING)
  add_executable(smac_tests tests/tests.cpp)
  target_link_libraries(smac_tests PRIVATE smac_formats)
  smac_set_warnings(smac_tests)
  add_test(NAME smac_tests COMMAND smac_tests)
endif()

if(SMAC_ENABLE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  foreach(target smac_core smac_formats smac-tool smac-native)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endforeach()
  if(TARGET smac_tests)
    target_compile_options(smac_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(smac_tests PRIVATE -fsanitize=address,undefined)
  endif()
endif()
